function mergeSession(session,headers){
 const values=typeof headers.getSetCookie==='function'?headers.getSetCookie():[headers.get('Set-Cookie')||''];
 const pairs=values.flatMap(v=>v.split(/,(?=\s*[^\s;,]+=)/)).map(v=>v.split(';')[0]);
 const merged={};for(const p of [...session.split(';'),...pairs]){const at=p.indexOf('=');if(at>0)merged[p.slice(0,at).trim()]=p.slice(at+1).trim();}
